        // State Management
        let savedPrograms = JSON.parse(localStorage.getItem('fahamu_saved')) || [];
        let currentCategory = 'all';

        // Filter Function
        function filterPrograms() {
            const query = document.getElementById('searchInput').value.toLowerCase();
            const ageVal = document.getElementById('ageFilter').value;
            const gradeVal = document.getElementById('gradeFilter').value;

            const filtered = programs.filter(p => {
                // Category Filter
                let matchesCategory = true;
                if (currentCategory === 'saved') matchesCategory = savedPrograms.includes(p.title);
                else if (currentCategory === 'free') matchesCategory = p.free === true;
                else if (currentCategory === 'quick') matchesCategory = p.quick === true;
                else if (currentCategory === 'local') matchesCategory = p.local === true;
                else if (currentCategory !== 'all') matchesCategory = p.category === currentCategory;

                // Age Filter
                let matchesAge = true;
                if (ageVal !== 'all') matchesAge = p.ageGroup === ageVal;

                // Grade Filter
                let matchesGrade = true;
                if (gradeVal !== 'all') matchesGrade = p.gradeGroup === gradeVal;

                // Search Query
                let matchesSearch = p.title.toLowerCase().includes(query) || 
                                    p.desc.toLowerCase().includes(query) || 
                                    p.tags.some(t => t.toLowerCase().includes(query));

                return matchesCategory && matchesAge && matchesGrade && matchesSearch;
            });

            renderPrograms(filtered);
        }

        function setCategory(cat, btnElement) {
            currentCategory = cat;
            
            // Reset active button styling
            document.querySelectorAll('#categoryContainer button').forEach(btn => {
                btn.className = "cat-btn bg-white text-gray-700 border border-gray-300 px-3 py-1.5 rounded-full";
            });
            if(btnElement) {
                btnElement.className = "cat-btn bg-navy text-white px-3 py-1.5 rounded-full shadow-sm";
            }

            filterPrograms();
        }

        function toggleSave(index) {
            const title = programs[index].title;
            if (savedPrograms.includes(title)) {
                savedPrograms = savedPrograms.filter(t => t !== title);
            } else {
                savedPrograms.push(title);
            }
            localStorage.setItem('fahamu_saved', JSON.stringify(savedPrograms));
            filterPrograms();
        }

        // Render Cards Grid
        function renderPrograms(list) {
            const grid = document.getElementById('cardsGrid');
            grid.innerHTML = '';

            if (list.length === 0) {
                grid.innerHTML = `<div class="col-span-full text-center py-12 text-gray-500 font-medium text-sm">No matching programs found. Try clearing your search or changing filters.</div>`;
                return;
            }

            list.forEach(p => {
                const originalIndex = programs.findIndex(orig => orig.title === p.title);
                const isSaved = savedPrograms.includes(p.title);

                const card = document.createElement('div');
                card.className = "bg-white rounded-xl shadow-md hover:shadow-xl transition p-5 flex flex-col justify-between border border-gray-100 relative";

                let tagsHTML = p.tags.map(t => `<span class="text-[10px] bg-slate-100 text-slate-700 px-2 py-0.5 rounded font-bold uppercase tracking-wider">${t}</span>`).join(' ');

                let badges = '';
                if (p.free) badges += '<span class="bg-emerald-100 text-emerald-800 text-[10px] font-black px-2 py-0.5 rounded-full mr-1">💚 FREE</span>';
                if (p.quick) badges += '<span class="bg-purple-100 text-purple-800 text-[10px] font-black px-2 py-0.5 rounded-full mr-1">⚡ QUICK</span>';
                if (p.local) badges += '<span class="bg-amber-100 text-amber-900 text-[10px] font-black px-2 py-0.5 rounded-full">🇰🇪 KENYA</span>';

                card.innerHTML = `
                    <div>
                        <div class="flex justify-between items-start gap-2 mb-2">
                            <h4 class="text-lg font-bold text-navy leading-snug pr-6">${p.title}</h4>
                            <button onclick="toggleSave(${originalIndex})" class="absolute top-5 right-5 text-xl transition transform hover:scale-110 focus:outline-none" title="Save for later">
                                ${isSaved ? '⭐' : '☆'}
                            </button>
                        </div>

                        <!-- Requirements Pills -->
                        <div class="flex flex-wrap items-center gap-1.5 text-[11px] text-gray-600 mb-3">
                            <span class="bg-slate-100 text-slate-800 px-2 py-0.5 rounded font-semibold">🎂 ${p.age}</span>
                            <span class="bg-slate-100 text-slate-800 px-2 py-0.5 rounded font-semibold">🎓 ${p.grade}</span>
                        </div>

                        <div class="mb-2">${badges}</div>
                        <div class="flex flex-wrap gap-1 mb-3">${tagsHTML}</div>
                        <p class="text-gray-600 text-xs leading-relaxed mb-4">${p.desc}</p>
                    </div>

                    <div class="space-y-2 pt-2 border-t border-gray-100">
                        <button onclick="openModal(${originalIndex})" class="w-full bg-slate-100 text-navy text-xs font-bold py-2.5 rounded-lg hover:bg-slate-200 transition text-center">
                            🔍 View Details & Requirements
                        </button>
                        <a href="${p.link}" target="_blank" rel="noopener noreferrer" class="block w-full bg-slate-900 text-white text-xs font-bold py-2.5 rounded-lg hover:bg-amber hover:text-navy transition text-center shadow-sm">
                            Visit & Apply &rarr;
                        </a>
                    </div>
                `;
                grid.appendChild(card);
            });
        }

        // Modal Functions
        function openModal(index) {
            const p = programs[index];
            document.getElementById('modalTitle').innerText = p.title;
            document.getElementById('modalAge').innerText = p.age + " years old";
            document.getElementById('modalGrade').innerText = p.grade;
            document.getElementById('modalDeadline').innerText = p.deadline;
            document.getElementById('modalFee').innerText = p.free ? "100% Free / Fully Funded" : "Paid (Financial Aid / Scholarships Available)";
            document.getElementById('modalDesc').innerText = p.desc;
            document.getElementById('modalTip').innerText = p.tip;
            document.getElementById('modalApplyBtn').href = p.link;
            document.getElementById('modalImg').style.backgroundImage = `url('${p.img}')`;
            document.getElementById('modalCategoryBadge').innerText = p.category.toUpperCase();

            document.getElementById('detailModal').classList.remove('hidden');
        }

        function closeModal() {
            document.getElementById('detailModal').classList.add('hidden');
        }

        // Close Modal when clicking background overlay
        document.getElementById('detailModal').addEventListener('click', function(e) {
            if (e.target === this) closeModal();
        });

        // Initial Page Render
        renderPrograms(programs);
    </script>
